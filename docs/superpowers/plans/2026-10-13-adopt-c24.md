# Adopt MetalUI C24 (resign focus on press) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Adopt MetalUI C24 for gap CT-a: put `.resignsFocusOnPress()` on the app's root so a primary press on inert chrome ends typing, and delete the shell's own press-resigns-focus stopgap (`AppModel.releaseTextFocus`, `window.focus(nil)`).

**Architecture:** MetalUI C24 (merged to `../MetalUI` master `0fb2ea6e`, PR #74, record section 98, rulings RF-A to RF-I, no divergence added) adds `.resignsFocusOnPress(_ isEnabled: Bool = true)` on any element group. It makes the group a resign region: a primary press inside it clears a focused field and routes no key by hover (unlike `.hoverKeyRegion()`); a press on a text field or a click-focusable element moves focus straight to it (A to B, no nil in between); a press on inert chrome resigns; `allowsHitTesting(false)`, `.disabled` and `.hidden()` withdraw the region; a `Deferred` presentation's press does not resign through a region below it; a secondary or other press resigns nothing; `@FocusState` observes the resign. The canvas editor (`CommentEditorField`) and the inspector's number and text fields already commit from `onChange(of: isFocused)` on focus loss, so the root modifier alone makes a press on glass padding, header gaps and inspector empty space end an edit as one undo step. No model code changes except deleting the hook.

**Tech Stack:** Swift 6.2, MetalUI (sibling `../MetalUI`), MetalUITesting headless `TestWindow` through the existing `AppWindow` fixture, Swift Testing.

**Spec:** `docs/metalui-gaps.md` entry CT-a (binding); `docs/superpowers/specs/2026-10-09-selection-groups-comments-design.md` section 8 (typing on the canvas: an edit ends "on a press elsewhere"); parent `docs/superpowers/specs/2026-10-07-metalcreator-vertical-slice-design.md` with all its Errata. import Testing

/// A place in `AppWindow`'s default layout (graph panel docked at the bottom here (corrected 2026-10-30: the default dock is left; these spots assume the bottom dock set explicitly), a comment selected or not) where a press
/// lands on no control: the human check CT-11's three spots. Each is a window point.
enum InertSpot: CaseIterable, CustomTestStringConvertible {
    /// The graph panel's glass padding, a few points inside its corner.
    case panelGlass
    /// The gap in the graph panel's header between the breadcrumbs and the Add button.
    case headerGap
    /// An empty area of the inspector: to the right of its first section title ("NOTE", "FRAME"), above the field.
    case inspectorEmptyArea

    var testDescription: String {
        switch self {
        case .panelGlass: "panel glass"
        case .headerGap: "header gap"
        case .inspectorEmptyArea: "inspector empty area"
        }
    }

    /// The spot's window point. The panel's frame is the one the panel measured.
    @MainActor func point(in app: AppWindow) throws -> Point<Pixels> {
        switch self {
        case .panelGlass:
            let panel = try #require(app.model.panelPlacement).panel
            return app.windowPoint(panel.origin + Vector2(4, 4))
        case .headerGap:
            let panel = try #require(app.model.panelPlacement).panel
            return app.windowPoint(panel.origin + Vector2(panel.size.x / 2, 24))
        case .inspectorEmptyArea:
            return app.windowPoint(Vector2(app.size.x - 70, 84))
        }
    }
}

import CreatorGeometry

/// A press that lands on nothing ends a canvas edit (spec 2026-10-09 §8: "on a press elsewhere"; gap CT-a, adopted from
/// MetalUI C24): `.resignsFocusOnPress()` on the window's root clears the field's focus, and the field's focus-loss
/// commit writes the draft as one step. Every press goes through the app's real window; the focus-loss commit runs in
/// the frame after the press, so each test runs `runUntilIdle()`.
@MainActor
struct CanvasResignOnPressWindowTests {
    let fixture = TypingFixture()

    func open() async throws -> AppWindow {
        try await AppWindow.open(fixture.graph)
    }

    /// The inspector's multi-line note box.
    func inspectorBox(_ app: AppWindow) throws -> TestElement {
        try #require(app.window.elements { $0.role == .textArea && $0.frame.origin.x.value >= Float(app.inspectorColumnStart) }.first)
    }

    @Test(arguments: InertSpot.allCases)
    func aPressOnInertChromeEndsANoteEditInOneStep(spot: InertSpot) async throws {
        let app = try await open()
        app.window.doubleClick(at: app.centre(ofComment: fixture.note.id))
        app.window.typeText("Kept ")
        app.window.click(at: try spot.point(in: app))
        try await app.window.runUntilIdle()
        #expect(app.editor.commentEdit == nil)
        #expect(app.editor.graph.stickies[fixture.note.id]?.text.hasSuffix("Kept ") == true)
        #expect(try app.window.focusedElement == nil)
        #expect(app.editor.document.undoName == "Edit Note")
        app.editor.document.undo()
        #expect(app.editor.graph.stickies[fixture.note.id]?.text == "Line one", "one step wrote the whole draft")
        #expect(!app.editor.document.canUndo)
    }

    @Test(arguments: InertSpot.allCases)
    func aPressOnInertChromeEndsAFrameTitleEditInOneStep(spot: InertSpot) async throws {
        let app = try await open()
        app.window.doubleClick(at: app.titleBarCentre(ofFrame: fixture.box.id))
        app.window.typeText("Two")  // the title opens selected, so typing replaces it
        app.window.click(at: try spot.point(in: app))
        try await app.window.runUntilIdle()
        #expect(app.editor.commentEdit == nil)
        #expect(app.editor.graph.frames[fixture.box.id]?.title == "Two")
        #expect(app.editor.document.undoName == "Edit Frame")
        app.editor.document.undo()
        #expect(app.editor.graph.frames[fixture.box.id]?.title == "Bracket")
        #expect(!app.editor.document.canUndo)
    }

    /// Human check CT-2's viewport press: the viewport is under the region too (the shell no longer clears the focus itself).
    @Test func aPressOnTheViewportEndsTheEditInOneStep() async throws {
        let app = try await open()
        app.window.doubleClick(at: app.centre(ofComment: fixture.note.id))
        app.window.typeText("Vp ")
        app.window.click(at: Point(x: Pixels(700), y: Pixels(300)))
        try await app.window.runUntilIdle()
        #expect(app.editor.commentEdit == nil)
        #expect(app.editor.graph.stickies[fixture.note.id]?.text.hasSuffix("Vp ") == true)
        #expect(try app.window.focusedElement == nil)
        app.editor.document.undo()
        #expect(app.editor.graph.stickies[fixture.note.id]?.text == "Line one" && !app.editor.document.canUndo)
    }

    /// A to B: the press lands on another field, the focus moves straight there, and the first edit commits once.
    @Test func aPressIntoTheInspectorsBoxMovesFocusAndCommitsTheCanvasEditOnce() async throws {
        let app = try await open()
        app.editor.canvasSelection = CanvasSelection(comments: [fixture.note.id])
        try await app.window.runUntilIdle()
        let box = try inspectorBox(app)
        app.window.doubleClick(at: app.centre(ofComment: fixture.note.id))
        app.window.typeText("Canvas ")
        try app.window.click(box)

    @Test func aDoubleClickOnAnotherCommentCommitsTheFirstOnce() async throws {

import CreatorGeometry
import CreatorGraph
import CreatorNodes
import MetalUI
import MetalUITesting
import Testing
@testable import CreatorApp
@testable import CreatorEditor

/// A number typed into an inspector field commits when a press elsewhere takes the focus (human check M6-6; gap CT-a,
/// adopted from MetalUI C24): the field's focus-loss commit, with the press landing on inert chrome or on another field.
/// The commit runs in the frame after the press, so each test runs `runUntilIdle()`.
@MainActor
struct InspectorFieldResignWindowTests {
    /// The window showing a Rectangle node, selected, and its inspector's number fields (Width, then Height).
    func selectedRectangle() async throws -> (app: AppWindow, node: Node, fields: [TestElement]) {
        var builder = GraphBuilder()
        let node = builder.add(RectangleNode.self, at: Vector2(40, 40))
        let app = try await AppWindow.open(builder.graph)
        app.window.click(at: app.point(inCanvas: Vector2(60, 80)))
        #expect(app.editor.selection == [node.id])
        let fields = try app.window.elements { $0.role == .textField && $0.frame.origin.x.value >= Float(app.inspectorColumnStart) }
        #expect(fields.count >= 2, "Width and Height")
        return (app, node, fields)
    }

    /// Selects a field's text and types `text` over it.
    func type(_ text: String, into field: TestElement, in app: AppWindow) throws {
        try app.window.click(field)
        app.window.typeKey("a", modifierFlags: .primary)
        app.window.typeText(text)
    }

    @Test(arguments: InertSpot.allCases)
    func aPressOnInertChromeCommitsATypedValueInOneStep(spot: InertSpot) async throws {
        let (app, node, fields) = try await selectedRectangle()
        try type("33", into: fields[0], in: app)
        #expect(app.editor.graph.nodes[node.id]?.inputValues["width"] == nil, "nothing is committed while typing")
        app.window.click(at: try spot.point(in: app))
        try await app.window.runUntilIdle()
        #expect(app.editor.graph.nodes[node.id]?.inputValues["width"] == .number(33))
        #expect(app.editor.document.undoName == "Change Width")
        #expect(try app.window.focusedElement == nil)
        app.editor.document.undo()
        #expect(app.editor.graph.nodes[node.id]?.inputValues["width"] == nil && !app.editor.document.canUndo, "one step")
    }

    /// A viewport press commits the typed value (a model commit, `AppModel+Viewport`) and takes the focus off the field.
    @Test func aPressOnTheViewportCommitsATypedValueAndClearsTheFocus() async throws {
        let (app, node, fields) = try await selectedRectangle()
        try type("33", into: fields[0], in: app)
        app.window.click(at: Point(x: Pixels(700), y: Pixels(300)))
        try await app.window.runUntilIdle()
        #expect(app.editor.graph.nodes[node.id]?.inputValues["width"] == .number(33))
        #expect(try app.window.focusedElement == nil)
        app.editor.document.undo()
        #expect(app.editor.graph.nodes[node.id]?.inputValues["width"] == nil && !app.editor.document.canUndo, "one step")
    }

    /// A to B: the press lands on Height, so Width commits (once) and Height has the focus.
    @Test func aPressIntoAnotherFieldCommitsTheFirstOnceAndMovesTheFocus() async throws {
        let (app, node, fields) = try await selectedRectangle()
        try type("33", into: fields[0], in: app)
        try app.window.click(fields[1])
        try await app.window.runUntilIdle()
        #expect(app.editor.graph.nodes[node.id]?.inputValues["width"] == .number(33))
        let focused = try #require(try app.window.focusedElement)
        #expect(focused.role == .textField && focused.frame.origin.y == fields[1].frame.origin.y, "Height has the focus")
        app.editor.document.undo()
        #expect(app.editor.graph.nodes[node.id]?.inputValues["width"] == nil && !app.editor.document.canUndo, "committed once")
    }

    /// A secondary press resigns nothing: the typed value is still pending, and typing carries on.
    @Test func aSecondaryPressOnInertChromeLeavesATypedValuePending() async throws {
        let (app, node, fields) = try await selectedRectangle()
        try type("33", into: fields[0], in: app)
        app.window.rightClick(at: try InertSpot.panelGlass.point(in: app))
        try await app.window.runUntilIdle()
        #expect(app.editor.graph.nodes[node.id]?.inputValues["width"] == nil)
        #expect(try app.window.focusedElement?.role == .textField)
        #expect(app.editor.pendingEntry?.text == "33")
    }
}
