`Sources/CreatorGraph/Hashing/HashPadding.swift` (new):

```swift
/// The padding SHA-1 and SHA-256 share (FIPS 180-4 §5.1.1): a 1 bit, zeros, and the message's length in bits as a
/// big-endian 64-bit number, so the padded message is a whole number of 64-byte blocks.
enum HashPadding {
    /// The 64-byte block size of both hashes.
    static let blockSize = 64

    /// The last one or two blocks of a message of `totalCount` bytes whose bytes after the last whole block are `tail`
    /// (fewer than 64 of them): `tail`, the 0x80 byte, zeros, then the bit length.
    static func finalBlocks(tail: UnsafeRawBufferPointer, totalCount: Int) -> [UInt8] {
        var bytes = [UInt8](tail)
        bytes.append(0x80)
        while bytes.count % blockSize != blockSize - 8 {
            bytes.append(0)
        }
        let bitCount = UInt64(totalCount) &* 8
        for shift in stride(from: 56, through: 0, by: -8) {
            bytes.append(UInt8(truncatingIfNeeded: bitCount >> UInt64(shift)))
        }
        return bytes
    }

    /// `bytes` as 64 lowercase hexadecimal digits (or 40, for SHA-1's twenty bytes): two digits a byte.
    static func hex(_ bytes: [UInt8]) -> String {
        let digits = Array("0123456789abcdef")
        return String(bytes.flatMap { [digits[Int($0 >> 4)], digits[Int($0 & 0x0F)]] })
    }
}
```

```swift
import Foundation

/// SHA-256 (FIPS 180-4), in plain Swift so a file's image keys are the same on every platform without CryptoKit. Pinned by
/// the NIST test vectors and, on macOS, by equality with CryptoKit's output (`SHA256Tests`). It is a content address, never a
/// secret, and is not constant-time.
enum SHA256 {
    /// The 32-byte digest of `data`.
    static func hash(_ data: Data) -> [UInt8] {
        data.withUnsafeBytes { hash($0) }
    }

    /// The 32-byte digest of `bytes`.
    static func hash(_ bytes: UnsafeRawBufferPointer) -> [UInt8] {
        var state: [UInt32] = [
            0x6A09_E667, 0xBB67_AE85, 0x3C6E_F372, 0xA54F_F53A, 0x510E_527F, 0x9B05_688C, 0x1F83_D9AB, 0x5BE0_CD19,
        ]
        withUnsafeTemporaryAllocation(of: UInt32.self, capacity: 64) { schedule in
            let whole = bytes.count / HashPadding.blockSize
            for block in 0..<whole {
                compress(&state, UnsafeRawBufferPointer(rebasing: bytes[block * 64..<block * 64 + 64]), schedule)
            }
            let tail = UnsafeRawBufferPointer(rebasing: bytes[(whole * 64)...])
            HashPadding.finalBlocks(tail: tail, totalCount: bytes.count).withUnsafeBytes { padded in
                for block in 0..<padded.count / HashPadding.blockSize {
                    compress(&state, UnsafeRawBufferPointer(rebasing: padded[block * 64..<block * 64 + 64]), schedule)
                }
            }
        }
        return state.flatMap { word in (0..<4).map { UInt8(truncatingIfNeeded: word >> UInt32(24 - 8 * $0)) } }
    }

    /// The 64 digits of `data`'s digest.
    static func hex(of data: Data) -> String {
        HashPadding.hex(hash(data))
    }

    /// Mixes one 64-byte `block` into `state`; `schedule` is 64 words of scratch.
    private static func compress(_ state: inout [UInt32], _ block: UnsafeRawBufferPointer,
                                 _ schedule: UnsafeMutableBufferPointer<UInt32>) {
        for index in 0..<16 {
            schedule[index] = block.loadUnaligned(fromByteOffset: index * 4, as: UInt32.self).bigEndian
        }
        for index in 16..<64 {
            let early = schedule[index - 15], late = schedule[index - 2]
            let sigma0 = rotate(early, 7) ^ rotate(early, 18) ^ (early >> 3)
            let sigma1 = rotate(late, 17) ^ rotate(late, 19) ^ (late >> 10)
            schedule[index] = schedule[index - 16] &+ sigma0 &+ schedule[index - 7] &+ sigma1
        }
        var (a, b, c, d, e, f, g, h) = (state[0], state[1], state[2], state[3], state[4], state[5], state[6], state[7])
        for index in 0..<64 {
            let sum1 = rotate(e, 6) ^ rotate(e, 11) ^ rotate(e, 25)
            let choice = (e & f) ^ (~e & g)
            let first = h &+ sum1 &+ choice &+ constants[index] &+ schedule[index]
            let sum0 = rotate(a, 2) ^ rotate(a, 13) ^ rotate(a, 22)
            let majority = (a & b) ^ (a & c) ^ (b & c)
            let second = sum0 &+ majority
            (h, g, f, e, d, c, b, a) = (g, f, e, d &+ first, c, b, a, first &+ second)
        }
        for (index, word) in [a, b, c, d, e, f, g, h].enumerated() {
            state[index] = state[index] &+ word
        }
    }
```

```swift
import Foundation

/// SHA-1 (FIPS 180-4), in plain Swift. Used only to name a UUID (version 5, RFC 4122 §4.3), as `NodeID.scoped` does: a
/// deterministic identity, not a security measure. Pinned by the NIST test vectors and, on macOS, by equality with CryptoKit's
/// `Insecure.SHA1` (`SHA1Tests`).
enum SHA1 {
    /// The 20-byte digest of `bytes`.
    static func hash(_ bytes: [UInt8]) -> [UInt8] {
        bytes.withUnsafeBytes { hash($0) }
    }

    /// The 20-byte digest of `bytes`.
    static func hash(_ bytes: UnsafeRawBufferPointer) -> [UInt8] {
        var state: [UInt32] = [0x6745_2301, 0xEFCD_AB89, 0x98BA_DCFE, 0x1032_5476, 0xC3D2_E1F0]
        withUnsafeTemporaryAllocation(of: UInt32.self, capacity: 80) { schedule in
            let whole = bytes.count / HashPadding.blockSize
            for block in 0..<whole {
                compress(&state, UnsafeRawBufferPointer(rebasing: bytes[block * 64..<block * 64 + 64]), schedule)
            }
            let tail = UnsafeRawBufferPointer(rebasing: bytes[(whole * 64)...])
            HashPadding.finalBlocks(tail: tail, totalCount: bytes.count).withUnsafeBytes { padded in
                for block in 0..<padded.count / HashPadding.blockSize {
                    compress(&state, UnsafeRawBufferPointer(rebasing: padded[block * 64..<block * 64 + 64]), schedule)
                }
            }
        }
        return state.flatMap { word in (0..<4).map { UInt8(truncatingIfNeeded: word >> UInt32(24 - 8 * $0)) } }
    }

    /// Mixes one 64-byte `block` into `state`; `schedule` is 80 words of scratch.
    private static func compress(_ state: inout [UInt32], _ block: UnsafeRawBufferPointer,
                                 _ schedule: UnsafeMutableBufferPointer<UInt32>) {
        for index in 0..<16 {
            schedule[index] = block.loadUnaligned(fromByteOffset: index * 4, as: UInt32.self).bigEndian
        }
        for index in 16..<80 {
            schedule[index] = rotate(schedule[index - 3] ^ schedule[index - 8] ^ schedule[index - 14] ^ schedule[index - 16], 1)
        }
        var (a, b, c, d, e) = (state[0], state[1], state[2], state[3], state[4])
        for index in 0..<80 {
            let mix: UInt32, constant: UInt32
            switch index {
            case 0..<20: (mix, constant) = ((b & c) | (~b & d), 0x5A82_7999)
            case 20..<40: (mix, constant) = (b ^ c ^ d, 0x6ED9_EBA1)
            case 40..<60: (mix, constant) = ((b & c) | (b & d) | (c & d), 0x8F1B_BCDC)
            default: (mix, constant) = (b ^ c ^ d, 0xCA62_C1D6)
            }
            let next = rotate(a, 5) &+ mix &+ e &+ constant &+ schedule[index]
            (e, d, c, b, a) = (d, c, rotate(b, 30), a, next)
        }
        for (index, word) in [a, b, c, d, e].enumerated() {
            state[index] = state[index] &+ word
        }
    }

    private static func rotate(_ value: UInt32, _ count: UInt32) -> UInt32 {
        (value << count) | (value >> (32 - count))
    }
}
```

```swift
// Test fixture file: a PNG encoder in plain Swift, so the image tests can make pictures where CoreGraphics does not exist.
import Foundation

/// Writes 8-bit PNGs with no compression: the pixel rows go into "stored" deflate blocks. The files are larger than ImageIO's but
/// valid, which is all a fixture needs (`PNGWriterTests` decodes them with ImageIO on macOS and checks their chunks everywhere).
enum PNGWriter {
    /// The PNG colour types this writer makes.
    enum ColorType: UInt8 {
        case grey = 0
        case rgba = 6

        /// Bytes per pixel at 8 bits per sample.
        var bytesPerPixel: Int { self == .grey ? 1 : 4 }
    }

    /// A `width` × `height` picture of `color`; `pixel(x, y)` gives its samples (one for grey, four for RGBA), each 0…255.
    static func encode(width: Int, height: Int, color: ColorType, pixel: (Int, Int) -> [UInt8]) -> Data {
        var raw: [UInt8] = []
        raw.reserveCapacity(height * (1 + width * color.bytesPerPixel))
        for y in 0..<height {
            raw.append(0)  // Filter type 0: the row as it is.
            for x in 0..<width {
                raw.append(contentsOf: pixel(x, y))
            }
        }
        var header: [UInt8] = []
        for side in [UInt32(width), UInt32(height)] {
            header.append(contentsOf: bigEndian(side))
        }
        header.append(contentsOf: [8, color.rawValue, 0, 0, 0])
        var file = Data([0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A])
        file.append(chunk("IHDR", header))
        file.append(chunk("IDAT", zlibStored(raw)))
        file.append(chunk("IEND", []))
        return file
    }

    /// The CRC-32 of `data` that ends every PNG chunk.
    static func crc32(_ data: Data) -> UInt32 {
        var crc: UInt32 = 0xFFFF_FFFF
        for byte in data {
            crc ^= UInt32(byte)
            for _ in 0..<8 { crc = crc & 1 == 1 ? (crc >> 1) ^ 0xEDB8_8320 : crc >> 1 }
        }
        return ~crc
    }

    /// One chunk: its length, its four-letter type, its bytes and the CRC of type and bytes.
    static func chunk(_ type: String, _ bytes: [UInt8]) -> Data {
        var body = Data(type.utf8)
        body.append(contentsOf: bytes)
        var chunk = Data(bigEndian(UInt32(bytes.count)))
        chunk.append(body)
        chunk.append(contentsOf: bigEndian(crc32(body)))
        return chunk
    }

    /// `raw` as a zlib stream of stored blocks of at most 65,535 bytes, closed by the Adler-32 of `raw`.
    private static func zlibStored(_ raw: [UInt8]) -> [UInt8] {
        var stream: [UInt8] = [0x78, 0x01]
        var offset = 0
        repeat {
            let count = min(65_535, raw.count - offset)
            let isLast = offset + count == raw.count
            stream.append(isLast ? 1 : 0)
            stream.append(contentsOf: [UInt8(count & 0xFF), UInt8(count >> 8), UInt8(~count & 0xFF), UInt8((~count >> 8) & 0xFF)])
            stream.append(contentsOf: raw[offset..<offset + count])
            offset += count
        } while offset < raw.count
        stream.append(contentsOf: bigEndian(adler32(raw)))
        return stream
    }

    private static func adler32(_ bytes: [UInt8]) -> UInt32 {
        var low: UInt32 = 1, high: UInt32 = 0
        for byte in bytes {
            low = (low + UInt32(byte)) % 65_521
            high = (high + low) % 65_521
        }
        return high << 16 | low
    }

    private static func bigEndian(_ value: UInt32) -> [UInt8] {
        [24, 16, 8, 0].map { UInt8((value >> UInt32($0)) & 0xFF) }
    }
}
```

```swift
/// `ImageLimits.maxSide`, any type but PNG and JPEG, and anything ImageIO cannot make sense of, all before a pixel is
/// decoded; `decode` then converts in bands of about 32 MB and checks for cancellation between bands. Memory for a legal
/// 8,192 × 8,192 picture: the 128 MB result and one 32 MB band, plus, for a PNG, the whole decoded `CGImage` that `decodeOnce`
/// keeps (ImageIO would otherwise decode it again for every band): about 256 MB for 8-bit RGBA and 512 MB for 16-bit, so a
/// PNG peaks near 0.4 GB (8-bit) to 0.7 GB (16-bit). A JPEG is read band by band and needs no full-size bitmap.
///
/// Grey comes from the picture's own values: a monochrome file is read as it is; colour is converted to sRGB and then to
/// Rec. 709 luminance (0.2126 R + 0.7152 G + 0.0722 B) of the colour premultiplied by alpha, i.e. composited against black.

    /// The file formats that may be used.
    public enum Format: Sendable, Equatable {
        case png, jpeg

        /// The extension a file of this format is saved with.
        public var fileExtension: String { self == .png ? "png" : "jpg" }
    }

    /// What a header says.
    public struct Info: Sendable, Equatable {
        public let width: Int
        public let height: Int
        public let format: Format
    }

    /// How many bytes of 16-bit bitmap one band may take.
    static let defaultBandBytes = 32 * 1024 * 1024

    /// The format and size of `data` from its header, checked against the limits, without decoding any pixels.
```

```swift
import Foundation

/// `MetalCreator --smoke-test [seconds]` (CI, cross-platform spec §2.6): opens the real window over SDL's offscreen
/// renderers, lets it run, and reports whether it reached its first frame. The window's root calls ``frameReached()`` from
/// a `.task`, which MetalUI starts inside the first frame, so a run that never draws never sets it.
@MainActor

    /// How long a smoke test runs when no number of seconds is given.
    public nonisolated static let defaultSeconds = 5.0

    public let seconds: Double

    public init(seconds: Double = LaunchSmoke.defaultSeconds) {
        self.seconds = seconds
    }

    /// Called by the window's root once its first frame has been built.
    public func frameReached() {
        reachedFirstFrame = true

    /// Whether the run passed: it reached its first frame.
    public var passed: Bool { reachedFirstFrame }

        if reachedFirstFrame {
            "MetalCreator smoke test: reached the first frame, ran \(seconds.formatted(.number)) s."
        } else {
            "MetalCreator smoke test: no frame was drawn in \(seconds.formatted(.number)) s."

    }
}
```

```python
#!/usr/bin/env python3
"""gate_tests.py <condition> <file> <testFuncName>... : wraps each named test function (with its attributes and doc comments) in #if <condition>."""
import sys, re
cond, path, names = sys.argv[1], sys.argv[2], sys.argv[3:]
lines = open(path).read().split('\n')
for name in names:
    idx = next((i for i, l in enumerate(lines) if re.search(r'func ' + re.escape(name) + r'\b', l)), None)
    if idx is None:
        sys.exit(f'{path}: no test named {name}')
    indent = re.match(r'\s*', lines[idx]).group(0)
    start = idx
    while start > 0 and (lines[start - 1].strip().startswith(('@', '///', '//')) ):
        start -= 1
    if lines[start - 1].strip() == f'#if {cond}':
        continue
    end = next(i for i in range(idx, len(lines)) if lines[i] == indent + '}')
    lines[end + 1:end + 1] = [indent + '#endif']
    lines[start:start] = [indent + '#if ' + cond]
open(path, 'w').write('\n'.join(lines))
```

`gate_occt.py` (gates, per file, the members that name OpenCascade or a gated helper):

```python
#!/usr/bin/env python3
"""gate_occt.py files... : gate, with #if canImport(CreatorOCCT), every member of a test type that names OpenCascade
(OCCTKernel, OCCTShape, OCCTProfile, ...), and make the file's `import CreatorOCCT` conditional.
Members are found by indentation: a member runs from its first attribute or doc-comment line to the closing brace at its indent."""
import re, sys

MENTION = re.compile(r'OCCT')

def depth_change(line):
    """The change in (), [] and {} nesting a source line makes, ignoring string literals and // comments."""
    code = re.sub(r'"(?:[^"\\]|\\.)*"', '""', line)
    code = code.split('//')[0]
    return sum(code.count(c) for c in '([{') - sum(code.count(c) for c in ')]}')

DECL = re.compile(r'\b(func|var|let|struct|enum|class|actor|init|typealias|subscript)\b')

def members(lines, indent):
    """Yield (start, end) line indexes of the members declared at `indent` spaces: from the first attribute or doc-comment
    line to the line where the declaration's brackets close."""
    pad = ' ' * indent
    i, n = 0, len(lines)
    while i < n:
        line = lines[i]
        if not (line.startswith(pad) and not line.startswith(pad + ' ') and line.strip()) or line.strip().startswith(('}', '#')):
            i += 1
            continue
        start, depth, seen = i, 0, False
        while i < n:
            text = lines[i]
            code = re.sub(r'"(?:[^"\\]|\\.)*"', '""', text).split('//')[0]
            depth += depth_change(text)
            if not text.strip().startswith('//') and DECL.search(code):
                seen = True
            if seen and depth <= 0:
                break
            i += 1
        yield start, min(i, n - 1)
        i += 1

def process(path):
    text = open(path).read()
    lines = text.split('\n')
    # the first type's body indent is 4
    all_members = list(members(lines, 4))
    blocks = {m: '\n'.join(lines[m[0]:m[1] + 1]) for m in all_members}
    gated = {m for m in all_members if MENTION.search(blocks[m])}
    # A member that uses a gated stored property or helper is gated too, unless it declares that name itself.
    changed = True
    while changed:
        changed = False
        names = set()
        for m in gated:
            # the declaration line is the first non-attribute, non-comment line of the member
            decl = next((l for l in blocks[m].split('\n') if not l.strip().startswith('//') and DECL.search(l)), '')
            found = re.search(r'\b(?:func|let|var|struct|enum|class|typealias)\s+(\w+)', decl)
            if found and not blocks[m].lstrip().startswith('@Test'):
                names.add(found.group(1))
        for m in all_members:
            if m in gated:
                continue
            body = blocks[m]
            for name in names:
                uses = re.search(r'(?<![\w.])' + re.escape(name) + r'\b', body)
                declares = re.search(r'\b(?:let|var|func)\s+' + re.escape(name) + r'\b', body.split('\n', 3)[-1])
                if uses and not declares:
                    gated.add(m)
                    changed = True
                    break
    out_ranges = sorted(gated)
    # merge adjacent ranges separated only by blank lines
    merged = []
    for s, e in out_ranges:
        if merged and all(not lines[k].strip() for k in range(merged[-1][1] + 1, s)):
            merged[-1] = (merged[-1][0], e)
        else:
            merged.append((s, e))
    for s, e in reversed(merged):
        lines[e + 1:e + 1] = ['    #endif']
        lines[s:s] = ['    #if canImport(CreatorOCCT)']
    text = '\n'.join(lines)
    text = re.sub(r'^((?:@testable )?import CreatorOCCT)$', r'#if canImport(CreatorOCCT)\n\1\n#endif', text, flags=re.M)
    open(path, 'w').write(text)
    print(path, len(merged), 'gated ranges')

for p in sys.argv[1:]:
    process(p)
```

```bash
#!/bin/bash
# Builds and tests MetalCreator on Linux in Docker (OrbStack on this Mac runs aarch64 containers; CI runs the same script
# on x86_64 and aarch64 runners). Cross-platform spec 2.5, 2.6.
#
#   scripts/linux-test.sh [build|test|smoke|all]    default: all
#     build   swift build
#     test    swift test, judged by the pass rule below
#     smoke   the app's --self-test, then --smoke-test 5 (the real window over SDL's offscreen renderers: a first frame in 5 s)
#     all     the three, in that order
#   scripts/linux-test.sh clean                     removes this track's build volume
#
# Needs MetalUI checked out beside this repository (../MetalUI), which Package.swift's path dependency expects. The repository
# is mounted read-write and MetalUI read-only; build products live in a named volume, so macOS's .build is untouched and a
# rerun is incremental.
#
# Pass rule for `test` (spec 2.5): exit 0, no "recorded an issue" or "failed after" line, and EXPECTED_TEST_RUNS lines
# "Test run with" (one per test target that has tests on Linux; macOS has one more, CreatorOCCTTests).
#
# Environment: XP_TRACK names the track (default xp-1), which names the container and the volume, so a script run kills only
# its own container. EXPECTED_TEST_RUNS overrides the count (10 today).
set -euo pipefail

cd "$(dirname "$0")/.."
REPO=$PWD
if [ ! -d ../MetalUI ]; then
    echo "scripts/linux-test.sh: MetalUI must be checked out beside this repository (../MetalUI)." >&2
    exit 2
fi
METALUI=$(cd ../MetalUI && pwd -P)

TRACK=${XP_TRACK:-xp-1}
EXPECTED_TEST_RUNS=${EXPECTED_TEST_RUNS:-10}
BASE_IMAGE=metalui-portable:latest
IMAGE=metalcreator-linux:latest
NAME=$TRACK-linux
VOLUME=$TRACK-linux-build
LOG=${TMPDIR:-/tmp}/$TRACK-linux-test.log

step=${1:-all}
if [ "$step" = clean ]; then
    docker volume rm "$VOLUME" >/dev/null 2>&1 || true
    exit 0
fi

if ! docker image inspect "$BASE_IMAGE" >/dev/null 2>&1; then
    echo "Building $BASE_IMAGE from MetalUI's recipe (once; it builds SDL3 from source)..."
    docker build -t "$BASE_IMAGE" -f "$METALUI/Backends/SDL/linux/Dockerfile" "$METALUI/Backends/SDL"
fi
docker build -q -t "$IMAGE" --build-arg "BASE=$BASE_IMAGE" - < scripts/linux/Dockerfile >/dev/null

# Runs `$1` in the image, from the repository, with the build volume; stops only this track's own container.
in_container() {
    docker rm -f "$NAME" >/dev/null 2>&1 || true
    docker run --rm --name "$NAME" \
        -v "$REPO":/work/MetalCreator -v "$METALUI":/work/MetalUI:ro -v "$VOLUME":/build \
        "$IMAGE" bash -ec "$1"
}

do_build() {
    in_container 'swift build --scratch-path /build'
}

do_smoke() {
    in_container 'bin=$(swift build --scratch-path /build --show-bin-path)/MetalCreatorApp
        "$bin" --self-test
        "$bin" --smoke-test 5'
}

do_test() {
    local status=0
    in_container 'swift test --scratch-path /build' 2>&1 | tee "$LOG" || status=$?
    local clean runs tests
    clean=$(sed $'s/\x1b\\[[0-9;]*m//g' "$LOG")
    runs=$(grep -c "Test run with" <<<"$clean" || true)
    tests=$(grep "Test run with" <<<"$clean" | sed -E 's/.*Test run with ([0-9]+) tests.*/\1/' | awk '{ total += $1 } END { print total + 0 }')
    if [ "$status" -ne 0 ] || grep -qE "recorded an issue|failed after" <<<"$clean" || [ "$runs" -ne "$EXPECTED_TEST_RUNS" ]; then
        echo "Linux: FAIL (exit $status, $runs of $EXPECTED_TEST_RUNS 'Test run with' lines; log: $LOG)" >&2
        return 1
    fi
    echo "Linux: PASS ($runs 'Test run with' lines, ${tests:-0} tests; log: $LOG)"
}

case $step in
    build) do_build ;;
    test) do_test ;;
    smoke) do_smoke ;;
    all) do_build && do_test && do_smoke ;;
    *) echo "usage: scripts/linux-test.sh [build|test|smoke|all|clean]" >&2; exit 64 ;;
esac
```
